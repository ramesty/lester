def __dataprep(customers_file, mails_file):
    import os
    from transformers import pipeline
    import lester as ld
    os.environ["TOKENIZERS_PARALLELISM"] = "False"
    import warnings

    warnings.simplefilter(action='ignore', category=FutureWarning)

    target_countries = ['UK', 'DE', 'FR']
    sentiment_predictor = pipeline('sentiment-analysis', model='distilbert-base-uncased-finetuned-sst-2-english')

    def matches_usecase(text):
        return "complaint" in text or "bank account" in text

    def sanitize(text):
        return text.lower()

    # Load customer data
    customers_df = ld.read_csv(customers_file, header=None, names=['customer_id', 'customer_email', 'bank', 'country', 'level'], sep=',', parse_dates=False)
    
    # Filter target countries
    customers_df = customers_df.filter("country in @target_countries")
    
    # Create is_premium column
    customers_df = customers_df.project('is_premium', ['level'], lambda level: level == 'premium')
    
    # Select necessary columns
    customers_df = customers_df[['customer_email', 'bank', 'country', 'is_premium']]
    
    # Load mails data
    mails_df = ld.read_csv(mails_file, header=None, names=['mail_id', 'email', 'raw_date', 'mail_subject', 'mail_text'], sep=',', parse_dates=False)
    
    # Extract mail year
    mails_df = mails_df.project('mail_year', ['raw_date'], lambda raw_date: int(raw_date.split("-")[0]))
    
    # Filter mails from 2022 onwards
    mails_df = mails_df.filter("mail_year >= 2022")
    
    # Join customer data with mails data
    merged_df = ld.join(mails_df, customers_df, left_on='email', right_on='customer_email')
    
    # Sanitize mail subject
    merged_df = merged_df.project('title', ['mail_subject'], sanitize)
    
    # Sanitize mail text
    merged_df = merged_df.project('text', ['mail_text'], sanitize)
    
    # Analyze sentiment
    merged_df = merged_df.project('sentiment', ['mail_text'], lambda text: sentiment_predictor(text)[0]['label'].lower())
    
    # Select final columns
    result_df = merged_df[['title', 'text', 'bank', 'country', 'sentiment', 'is_premium']]
    
    return result_df

def __featurise():
    import numpy as np
    from sklearn.base import BaseEstimator, TransformerMixin
    from sklearn.preprocessing import StandardScaler, OneHotEncoder
    from sklearn.pipeline import Pipeline
    from sklearn.compose import ColumnTransformer
    from sentence_transformers import SentenceTransformer
    from sklearn.model_selection import train_test_split
    import pandas as pd

    class SentenceEmbeddingTransformer(BaseEstimator, TransformerMixin):
        def __init__(self, model_name='all-mpnet-base-v2'):
            self.model_name = model_name
            self.embedder = SentenceTransformer(model_name)

        def fit(self, X, y=None):
            return self

        def transform(self, X, y=None):
            return self.embedder.encode(X)

    class TitleLengthTransformer(BaseEstimator, TransformerMixin):
        def fit(self, X, y=None):
            return self

        def transform(self, X, y=None):
            title_lengths = np.array([len(title) for title in X])
            return title_lengths.reshape(-1, 1)

    # Load data
    df = pd.read_csv(".scratchspace/__intermediate.csv", sep="\t", header=None)
    df.columns = ['title', 'text', 'bank', 'country', 'sentiment', 'is_premium']

    # Define transformers
    sentence_embedding_transformer = SentenceEmbeddingTransformer()
    title_length_transformer = Pipeline([
        ('length', TitleLengthTransformer()),
        ('scaler', StandardScaler())
    ])
    country_onehot_transformer = OneHotEncoder(categories=[['DE', 'FR', 'UK']], drop=None)

    # Define column transformer
    preprocessor = ColumnTransformer(
        transformers=[
            ('title_embedding', sentence_embedding_transformer, 'title'),
            ('text_embedding', sentence_embedding_transformer, 'text'),
            ('title_length', title_length_transformer, 'title'),
            ('country_onehot', country_onehot_transformer, ['country'])
        ]
    )

    return preprocessor

def __model(num_features):
    import torch
    import torch.nn as nn

    class LogisticRegressionModel(nn.Module):
        def __init__(self, num_features):
            super(LogisticRegressionModel, self).__init__()
            self.linear = nn.Linear(num_features, 1)

        def forward(self, x):
            return torch.sigmoid(self.linear(x))

    model = LogisticRegressionModel(num_features)
    loss = nn.BCELoss()

    return model, loss
