def __featurise():
    import numpy as np
    from sklearn.preprocessing import StandardScaler, OneHotEncoder
    from sklearn.pipeline import Pipeline
    from sklearn.compose import ColumnTransformer
    from sentence_transformers import SentenceTransformer
    from sklearn.base import BaseEstimator, TransformerMixin
    import pandas as pd
    
    # Custom transformer for sentence embeddings
    class SentenceEmbeddingTransformer(BaseEstimator, TransformerMixin):
        def __init__(self, model_name="all-mpnet-base-v2"):
            self.model_name = model_name
            self.embedder = None
            
        def fit(self, X, y=None):
            self.embedder = SentenceTransformer(self.model_name)
            return self
        
        def transform(self, X):
            return self.embedder.encode(X)

    # Custom transformer for word count
    class WordCountTransformer(BaseEstimator, TransformerMixin):
        def fit(self, X, y=None):
            return self
        
        def transform(self, X):
            return np.array([len(text.split(" ")) for text in X]).reshape(-1, 1)

    # Define the transformers
    title_embedding_transformer = Pipeline(steps=[
        ('sentence_embedding', SentenceEmbeddingTransformer())
    ])
    
    text_embedding_transformer = Pipeline(steps=[
        ('sentence_embedding', SentenceEmbeddingTransformer())
    ])
    
    title_length_standardizer = Pipeline(steps=[
        ('word_count', WordCountTransformer()),
        ('scaler', StandardScaler())
    ])
    
    country_onehot_encoder = OneHotEncoder(categories=[['DE', 'FR', 'UK']], drop=None, sparse=False)
    
    # Column Transformer
    preprocessor = ColumnTransformer(
        transformers=[
            ('title_embedding', title_embedding_transformer, 'title'),
            ('text_embedding', text_embedding_transformer, 'text'),
            ('title_length', title_length_standardizer, 'title'),
            ('country_onehot', country_onehot_encoder, ['country'])
        ]
    )
    
    return preprocessor

            self.scaler_.fit(lengths)
            return self
        
        def transform(self, X):

            return self.scaler_.transform(lengths)

    country_encoder = OneHotEncoder(categories=[list(country_indices.keys())], handle_unknown='ignore')
    
    column_transformer = ColumnTransformer(
        transformers=[
            ('title_embedding', SentenceEmbeddingTransformer(), 'title'),
            ('text_embedding', SentenceEmbeddingTransformer(), 'text'),
            ('title_length', TitleLengthTransformer(), ['title']),
            ('country_onehot', country_encoder, ['country'])
        ],
        remainder='drop'
    )
    
    return column_transformer
